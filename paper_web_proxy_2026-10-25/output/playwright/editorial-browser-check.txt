### Ran Playwright code
```js
await (async (page) => {
 await page.setViewportSize({width:1280,height:1000});
 await page.goto('http://127.0.0.1:8320/webpage/dhc6_report.html', {waitUntil:'networkidle'});
 await page.evaluate(() => document.fonts.ready);
 const desktop = await page.evaluate(() => ({title:document.title, viewport:innerWidth, pageWidth:document.documentElement.scrollWidth, mathErrors:document.querySelectorAll('.katex-error').length, missingImages:[...document.images].filter(i=>!i.complete||!i.naturalWidth).length, method:document.getElementById('s3').textContent, results:document.getElementById('s5').textContent, processHeadings:[...document.querySelectorAll('h2,h3,h4')].filter(e=>/第[一二三四]步/.test(e.textContent)).length}));
 await page.screenshot({path:'editorial-abstract.png'});
 await page.setViewportSize({width:390,height:844});
 await page.screenshot({path:'editorial-abstract-mobile.png'});
 const mobile = await page.evaluate(() => ({viewport:innerWidth,pageWidth:document.documentElement.scrollWidth}));
 if(desktop.mathErrors||desktop.missingImages||desktop.processHeadings||desktop.pageWidth>desktop.viewport||mobile.pageWidth>mobile.viewport) throw Error(JSON.stringify({desktop,mobile}));
 console.log(JSON.stringify({desktop,mobile}));
})(page);
```
